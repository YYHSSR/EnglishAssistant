cmake_minimum_required(VERSION 3.10)
project(EnglishAssistant VERSION 0.7.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  add_library(assistant_core src/dictionary.cpp src/offline.cpp)
  target_include_directories(assistant_core PUBLIC src)
  target_compile_options(assistant_core PRIVATE -Wall -Wextra -Wpedantic)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
    target_link_libraries(assistant_core PUBLIC stdc++fs)
  endif()
  add_executable(EnglishAssistantCore src/core_cli.cpp)
  target_link_libraries(EnglishAssistantCore PRIVATE assistant_core)
  add_executable(OfflineTests tests/offline_tests.cpp)
  target_link_libraries(OfflineTests PRIVATE assistant_core)
  enable_testing()
  add_test(NAME offline COMMAND OfflineTests "${CMAKE_CURRENT_SOURCE_DIR}")
  return()
endif()
enable_language(RC)
add_library(assistant_core src/dictionary.cpp src/candidates.cpp src/options.cpp src/layout.cpp src/offline.cpp src/startup.cpp src/translation_box.cpp src/documents.cpp)
target_include_directories(assistant_core PUBLIC src)
target_compile_definitions(assistant_core PUBLIC UNICODE _UNICODE)
target_link_libraries(assistant_core PUBLIC uiautomationcore ole32 oleaut32 uuid user32 advapi32 comctl32 gdi32)
add_executable(EnglishAssistant WIN32 src/app.cpp resources/app.rc)
target_include_directories(EnglishAssistant PRIVATE resources)
target_link_libraries(EnglishAssistant PRIVATE assistant_core shell32 gdi32 dwmapi shcore)
enable_testing()
add_executable(CoreTests tests/core_tests.cpp)
target_link_libraries(CoreTests PRIVATE assistant_core)
add_executable(ReadCandidates tests/read_candidates.cpp)
target_link_libraries(ReadCandidates PRIVATE assistant_core)
add_test(NAME dictionary COMMAND CoreTests "${CMAKE_CURRENT_SOURCE_DIR}/data/glossary-en.tsv")
add_executable(OfflineTests tests/offline_tests.cpp)
target_link_libraries(OfflineTests PRIVATE assistant_core)
add_test(NAME offline COMMAND OfflineTests "${CMAKE_CURRENT_SOURCE_DIR}")
if(MINGW)
  target_compile_options(assistant_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(EnglishAssistant PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(EnglishAssistant PRIVATE -municode -static -s)
  target_link_options(CoreTests PRIVATE -static)
  target_link_options(ReadCandidates PRIVATE -static)
  target_link_options(OfflineTests PRIVATE -static)
endif()
