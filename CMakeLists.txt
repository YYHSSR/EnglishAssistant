cmake_minimum_required(VERSION 3.20)
project(PinyinShift VERSION 0.9.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "PinyinShift supports Windows only.")
endif()
enable_language(RC)
set_property(SOURCE resources/app.rc APPEND PROPERTY OBJECT_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/resources/app.ico"
  "${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest")
add_library(assistant_core src/dictionary.cpp src/candidates.cpp src/options.cpp src/layout.cpp src/offline.cpp src/startup.cpp src/translation_box.cpp src/documents.cpp src/neural.cpp src/sentence.cpp src/background.cpp)
target_include_directories(assistant_core PUBLIC src)
target_compile_definitions(assistant_core PUBLIC UNICODE _UNICODE)
target_link_libraries(assistant_core PUBLIC uiautomationcore ole32 oleaut32 uuid user32 advapi32 comctl32 gdi32 gdiplus mfplat mfreadwrite mfplay mfuuid comdlg32 shell32)
add_executable(PinyinShift WIN32 src/app.cpp resources/app.rc)
target_include_directories(PinyinShift PRIVATE resources)
target_link_libraries(PinyinShift PRIVATE assistant_core shell32 gdi32 dwmapi shcore)
enable_testing()
add_executable(CoreTests tests/core_tests.cpp)
target_link_libraries(CoreTests PRIVATE assistant_core)
add_executable(ReadCandidates tests/read_candidates.cpp)
target_link_libraries(ReadCandidates PRIVATE assistant_core)
add_test(NAME dictionary COMMAND CoreTests "${CMAKE_CURRENT_SOURCE_DIR}/data/glossary-en.tsv")
add_executable(OfflineTests tests/offline_tests.cpp)
target_link_libraries(OfflineTests PRIVATE assistant_core)
add_test(NAME offline COMMAND OfflineTests "${CMAKE_CURRENT_SOURCE_DIR}")
if(MINGW)
  target_compile_options(assistant_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(PinyinShift PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(PinyinShift PRIVATE -municode -static -s)
  target_link_options(CoreTests PRIVATE -static)
  target_link_options(ReadCandidates PRIVATE -static)
  target_link_options(OfflineTests PRIVATE -static)
endif()
add_executable(NeuralTests tests/neural_tests.cpp)
target_link_libraries(NeuralTests PRIVATE assistant_core)
if(MINGW)
  target_link_options(NeuralTests PRIVATE -static)
endif()
add_test(NAME neural COMMAND NeuralTests "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(WindowsUITests tests/windows_ui.cpp resources/app.rc)
target_include_directories(WindowsUITests PRIVATE resources)
target_link_libraries(WindowsUITests PRIVATE assistant_core)
if(MINGW)
  target_link_options(WindowsUITests PRIVATE -static)
endif()
add_test(NAME windows_ui COMMAND WindowsUITests "${CMAKE_CURRENT_SOURCE_DIR}")
